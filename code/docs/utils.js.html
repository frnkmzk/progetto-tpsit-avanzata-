/**
 * @function validaPassword
 * @description controlla se la password data come parametro rispetta le condizioni di sicurezza
 * @param {string} password password da verificare
 * @returns {boolean}
 */
export function validaPassword(password) {



let hasUpper = false;


let hasLower = false;


let hasNumber = false;


let hasMinLength = password.length >= 8;

  for (let i = 0; i < password.length; i++) {
      const char = password[i];

      if (char >= 'A' && char <= 'Z') {
          hasUpper = true;
      } else if (char >= 'a' && char <= 'z') {
          hasLower = true;
      } else if (char >= '0' && char <= '9') {
          hasNumber = true;
      }
  }

  return hasUpper && hasLower && hasNumber && hasMinLength;
}

/**
 * @function accessoAdmin
 * @description permette l'accesso alla mod admin
 * @param {object} cred credenziali admin
 * @returns {boolean} risultato verifica
 */
export function accessoAdmin(cred){

   let tentUser=Prompt("inserire user > : ")
   let tentPass=Prompt("inserire password > : ")

   return tentUser === cred.user && tentPass === cred.pass;
}

/**
 * @function trova
 * @description usa il metodo find che restituisce l'ogetto u una volta trovato l'username corrispondente
 * @param {string} user username da trovare
 * @param {Array} elenco elenco in cui cercarlo
 * @returns {object} u,oggetto trovato con user corrispondente || -1 di default se non trovato
 */
export function trova(user,elenco)
{
  return elenco.find(u => u.user === user)
}
